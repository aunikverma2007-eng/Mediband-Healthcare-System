  <script>
    let alarmInterval = null;
    let audioCtx = null;
    let currentLat = null;
    let currentLon = null;
    let mapsUrl = "https://www.google.com/maps/search/nearest+hospital";

    // Loud Emergency Siren using Web Audio API (100% Free, No external MP3 needed)
    function startSiren() {
      if (alarmInterval) return;
      audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      alarmInterval = setInterval(() => {
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(880, audioCtx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(440, audioCtx.currentTime + 0.4);
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.45);
      }, 600);
    }

    function stopSiren() {
      if (alarmInterval) {
        clearInterval(alarmInterval);
        alarmInterval = null;
      }
      if (audioCtx) {
        audioCtx.close();
        audioCtx = null;
      }
    }

    // Fetch Live Location & Auto-Find Nearest Hospital
    function detectLocationAndHospital(openMap = false) {
      const locText = document.getElementById('locationText');
      const hospText = document.getElementById('nearestHospitalText');

      if ("geolocation" in navigator) {
        locText.innerText = "Fetching live GPS coordinates...";
        navigator.geolocation.getCurrentPosition(
          (pos) => {
            currentLat = pos.coords.latitude.toFixed(5);
            currentLon = pos.coords.longitude.toFixed(5);
            mapsUrl = `https://www.google.com/maps/search/hospital/@${currentLat},${currentLon},15z`;
            locText.innerHTML = `Lat: ${currentLat}, Lng: ${currentLon} <a href="https://maps.google.com/?q=${currentLat},${currentLon}" target="_blank" class="underline text-red-600 ml-1">(View Map)</a>`;
            hospText.innerHTML = `Nearest Emergency Hospital Ready <a href="${mapsUrl}" target="_blank" class="underline text-red-600 font-extrabold ml-1">Open Route →</a>`;
            updateFamilyPreview(true);
            if (openMap) window.open(mapsUrl, '_blank');
          },
          () => {
            locText.innerText = "GPS Permission Blocked — Using City Emergency Zone";
            hospText.innerHTML = `<a href="${mapsUrl}" target="_blank" class="underline text-red-600 font-extrabold">Search Nearest Hospital on Google Maps →</a>`;
            if (openMap) window.open(mapsUrl, '_blank');
          }
        );
      }
    }

    function findNearestHospital() {
      detectLocationAndHospital(true);
    }

    async function generateFirstAid(reason) {
      const apiKey = document.getElementById('geminiKey').value.trim();
      const box = document.getElementById('aiFirstAidBox');
      const problem = document.getElementById('pProblemDisplay').innerText;

      if (!apiKey) {
        box.innerHTML = `
          <p class="font-bold text-red-600 mb-1">Immediate First-Aid Steps (Offline Mode):</p>
          <ul class="list-disc pl-5 space-y-1">
            <li>Patient ko seedha litayein aur bheed hata kar taazi hawa aane dein.</li>
            <li>Agar unconscious hain toh recovery position (ek karwat) mein litayein.</li>
            <li>Koi bhi paani ya dawai zabardasti muh mein na dalein.</li>
            <li>108 Ambulance aur Nearest Hospital route upar active kar diya gaya hai.</li>
          </ul>`;
        return;
      }

      box.innerHTML = `<p class="font-bold animate-pulse">Gemini AI emergency first-aid tayar kar raha hai...</p>`;
      try {
        const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            contents: [{ parts: [{ text: `Emergency situation: ${reason}. Patient medical history: ${problem}. Give 4 very simple, urgent first-aid bullet points in Hinglish for bystanders until the ambulance arrives.` }] }]
          })
        });
        const data = await res.json();
        const reply = data.candidates[0].content.parts[0].text.replace(/\*\*(.*?)\*\*/g, '<b>$1</b>').replace(/\n/g, '<br>');
        box.innerHTML = reply;
      } catch (e) {
        box.innerHTML = `<p class="text-red-600 font-bold">Could not reach Gemini AI. Follow standard recovery position & wait for 108 Ambulance.</p>`;
      }
    }
